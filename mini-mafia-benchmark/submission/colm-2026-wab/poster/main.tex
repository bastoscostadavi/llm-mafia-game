\sectionTitle{A Closed-Form Interaction Model}
\panel{%
  \begin{center}
  \colorbox{ciaamPale}{\parbox{0.88\linewidth}{
    \centering
    \vspace{0.10in}
    {\fontsize{26}{31}\selectfont\color{ciaamDark}
      $\operatorname{logit}(p_{ijk}) = v_k\,(m_i-d_j)$}
    \vspace{0.09in}
  }}
  \end{center}

  \begin{itemize}[leftmargin=0.22in,itemsep=0.04in,topsep=0.02in]
    \item Difference only: outcomes depend on $m_i-d_j$.
    \item Neutral point: $p_{ijk}=1/2$ when $m_i-d_j=0$ or $v_k=0$.
    \item Shift symmetry: $(m,d)\mapsto(m+c,d+c)$.
    \item Scale symmetry: $(v,m,d)\mapsto(v/\lambda,\lambda m,\lambda d)$.
    \item Gauge fixing: $\mathbb{E}[m]=0$ and $\mathbb{E}[v]=1$.
  \end{itemize}
}

\sectionTitle{Why Parameterize? + Fit Diagnostics}
\panel{%
  \begin{itemize}[leftmargin=0.22in,itemsep=0.04in,topsep=0.02in]
    \item \textbf{Prediction:} infer unseen role combinations.
    \item \textbf{Diagnosis:} separate deception, disclosure, and detection failures.
    \item \textbf{Comparison:} expose role specialization across models.
    \item \textbf{Reliable fit:} max $\hat{R}=1.007$; bulk ESS $>360$; tail ESS $>430$.
  \end{itemize}
}

\sectionTitle{Discussion}
\panel{%
  \begin{itemize}[leftmargin=0.22in,itemsep=0.075in,topsep=0.02in]
    \item \textbf{Parameterization:} $3I$ transferable capabilities summarize $I^3$ outcomes, predict unseen combinations, and expose interaction-level effects.
    \item \textbf{Future directions:} test richer roles, larger casts, longer horizons, broader model families, and explicit belief updates.
    \item \textbf{AI safety:} labeled deceptive and truthful statements support deception detection, capability tracking, and analysis of strategic-reasoning failures.
  \end{itemize}
}
